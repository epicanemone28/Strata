# Kana tables for Katakana.java: the letter-to-kana rules, letter names and number words.
# This file is required: without it the chat voice has nothing to say.
#
# Lines starting with # and blank lines are ignored. Everything is UTF-8.

[rows]
# Consonant = the kana for that consonant + a i u e o (five entries). "-" is no consonant.
# Keys the converter looks up: - k g s z t d n h f b p m y r w v j sh ch q

k  = カ キ ク ケ コ
g  = ガ ギ グ ゲ ゴ

p  = パ ピ プ ペ ポ
m  = マ ミ ム メ モ
y  = ヤ イ ユ イェ ヨ
r  = ラ リ ル レ ロ
